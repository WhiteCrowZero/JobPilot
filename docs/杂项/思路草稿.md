## 仍不明白

为什么 repository 这些都要写成`self, *`，但明白有使用self方法
比如：

```python
class UserSkillRepository:
    """用户技能画像数据库操作。"""

    async def skill_exists(self, db: AsyncSession, *, skill_id: int) -> bool:
        """判断标准技能是否存在。"""

        stmt = select(Skill.id).where(Skill.id == skill_id).limit(1)
        return await db.scalar(stmt) is not None
```

为什么一些一句话的函数（并且只使用一次）要单独抽出来，而不是直接写在大函数里面
比如：

```python
class UserSkillService:

    def __init__(self, repository: UserSkillRepository) -> None:
        self.repository = repository

    async def upsert_user_skill(
            self,
            db: AsyncSession,
            *,
            user_id: int,
            payload: UserSkillUpsert,
    ) -> UserSkillResponse:
        """新增、更新或恢复当前用户技能画像。"""

        try:
            if not await self.repository.skill_exists(db, skill_id=payload.skill_id):
                raise StandardSkillNotFoundError()
            user_skill = await self.repository.upsert_active_skill(
                db,
                user_id=user_id,
                skill_id=payload.skill_id,
                create_values=self._build_upsert_create_values(payload),
                update_values=self._build_upsert_update_values(payload),
            )
            await db.commit()
        except Exception:
            await db.rollback()
            raise
        return self._to_response(user_skill)

    @staticmethod
    def _build_upsert_create_values(payload: UserSkillUpsert) -> dict[str, object]:
        return payload.model_dump(exclude={"skill_id"})

    @staticmethod
    def _build_upsert_update_values(payload: UserSkillUpsert) -> dict[str, object]:
        return payload.model_dump(exclude={"skill_id"}, exclude_unset=True)
```
